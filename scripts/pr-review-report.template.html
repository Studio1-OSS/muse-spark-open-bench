The cases, grades, reproduction scripts, every review and the per-request token ledger ship with the repository.

# build the 31 case folders from local clones
node --import tsx scripts/build-review-cases.ts REPOS_DIR
# run the bugs Muse described on before/merged/fix
node --import tsx scripts/pr-review-repro.ts REPOS_DIR
# review every PR with Muse Spark Open
npm run bench -- run
# rebuild this page from the results
node --import tsx scripts/pr-review-report.ts
